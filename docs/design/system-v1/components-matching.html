06 · RN API (proposed)

Component contracts

<Button
  variant="cta" | "primary" | "secondary" | "ghost" | "destructive"
  size="sm" | "md" | "lg"        // 36 / 48 / 52
  state="default" | "loading" | "disabled"
  iconLeft? icon? iconRight?     // optional 18pt icons
  haptic="light" | "medium" | "success" | "none"  // default light
  full?                          // stretch to container
  onPress
/>

<ActionButton
  action="pass" | "bless" | "super"
  onPress
  // size + halo + haptic preset by action
/>

<Sheet
  variant="half" | "full" | "confirm" | "match" | "pastor"
  visible
  onDismiss
  detents?: ['half', 'full']     // half/full only
  dismissible? = variant !== "pastor" && variant !== "match"
  eyebrow? title? children
  primaryAction?: ButtonProps
  secondaryAction?: ButtonProps
/>
